### 3.1 实测命令（可复制，UTF-8 终端）

ATTR = re.compile(r'(?:href|src)\s*=\s*"([^"]+)"', re.I)
def is_rel(v):
    v = v.strip()
    if not v: return False
    if re.match(r'^(https?:|mailto:|javascript:|data:|tel:|#|//)', v): return False

for p in htmls:
